## 恢复结构、ABI与正常流程

下列为恢复伪代码，非原源码。RV32 word/pointer 均4 B；`Header` 长12 B。

```c
struct Registry { uint32_t size; Decoder *head, *tail; }; // G+110, +114, +118
struct Decoder {
    InfoFn info;             // +00
    OpenFn open;             // +04
    uint32_t field08;        // +08, PNG注册后仍0
    CloseFn close;           // +0c
    uint32_t field10;        // +10, 0
    const char *name;        // +14
    uint32_t field18;        // +18, 0
    Decoder *prev, *next;    // +1c/+20，尾置链接，合计36 B
};
struct Header {
    uint8_t byte0, cf; uint16_t flags; // +0/+1/+2
    uint16_t width, height, stride, reserved; // +4/+6/+8/+a
};

// 仅表示本次已知正常global注册片段；省略其它GUI初始化。
zero(G, 0x210); L.size = 28;
bin = create();                    // prepend + clear first28
install_BIN_callbacks(bin);
png = create();                    // prepend，所以png -> bin
png->info = a00eb6fe;
png->open = a00eb7b6;
png->close = a00eab9a;
png->name = (char*)a02634f0;
```

create 正常分配成功后返回node；分配失败经过原有log/assert路径，随后仍可达payload清零代码。本轮没有执行失败路径，也不擅自替原程序添加 `return NULL`；所有正常注册和存续结论以分配成功为前提。node先发布到list，再由create清payload和注册者写callback，不由此推断并发初始化保证。

```c
// local共76 B，由wrapper清零；source在+c，class byte在+10。
bool wrapper(void *source, Header *out) {
    byte local[76] = {0};
    store32(local+12, source);
    local[16] = classify(source);   // 12个源0x19 -> class0，沿用A20
    return dispatcher(local,out) != NULL;
}

// class0简化路径；common cache元数据读取不可视作不存在。
Decoder *class0_info(Local *d, Header *out) {
    zero(out,12);
    if (d->source->data == NULL) return NULL;
    cache = load32(G+0x120);
    metadata = load32(cache+8);     // 公共入口读取，class0不用它作cache查找
    for (Decoder *n=L.head; n; n=load32((byte*)n+L.size+4)) {
        if (!n->info || !n->open) continue;
        a0089bde((byte*)d+0x14,0,0); // return ignored
        InfoFn f = n->info;         // helper之后重读
        if (f(n,d,out) != 1) continue;
        if (out->stride == 0) {
            uint32_t w=out->width;
            out->stride = (out->cf==0x14)
                ? (uint16_t)(w<<1)
                : (uint16_t)((w*depth(out->cf)+7)>>3);
        }
        return n;
    }
    return NULL;
}
```

`a0089bde` 先 `9be8` 读取参数+4，NULL经 `9bee` 跳 `9c26`返回，a0预置11。对wrapper清零后的local，参数local+14的+4即local+18为0，所以PNG首节点第一次info调用前，此helper没有调用其非空状态的间接callback，也没有写该local。A21把它留作明确接口不会改变结论；本轮只补核该必要空状态门，未恢复整个文件接口。

```c
int png_class0_info(Decoder *unused, Local *d, Header *out) {
    Descriptor *s=d->source;
    if (s->size<=7u) return 0;
    if (compare(s->data, rom_signature, 8)!=0) return 0;
    out->cf=0x10;
    uint32_t x=load_le32(s->data+16);
    out->width=(uint16_t)(((x>>8)&0xffffff00u)+(x>>24));
    x=load_le32(s->data+20);
    out->height=(uint16_t)(((x>>8)&0xffffff00u)+(x>>24));
    return 1;
}
```

共同输出片段实际是 LW→SRLI8→ANDI -256→SRLI24→ADD→SH，结果为PNG大端尺寸的低16位。函数class>1回0；class1有读24 B/count==24/signature门并共享输出，文件依赖保持接口边界。这里没有执行PNG chunk/CRC/解压；大小门只按原件记录，不扩大异常输入测试。

12个既有源条件计算全部吻合：category1为19×19、stride76、header `00100000130013004c000000`；category3为20×16、stride80、header `001000001400100050000000`；其余10源20×20、stride80、header `001000001400140050000000`。每个源地址、原header、data指针、data尺寸字节及条件结果见本轮JSON。源首4 B `19010000` 与输出首4 B `00100000` 明确区分。

这些结果还要求：正常global初始化完成且对象有效，分配成功，已知list关系未被后续注册/改写遮蔽，ROM数据不变，调用期间未被清理。由此静态推断PNG节点首个接受、dispatcher返回该节点、wrapper为1；它不是运行观测。A20 setter只有info成功及aux guard通过才更新source/尺寸/格式和请求刷新，本轮不外推guard或最终显示必然发生。

## 清理、状态与所有权

`a008be06` 入口先置 `G+1=1`（`be2e`），清理其它global对象；它不是只有registry的孤立析构。此前业务callee仅留边界。cache销毁后到达本registry clear时：

```c
void clear_registry(Registry *l) {
    Decoder *n=l->head;
    while (n) {
        Decoder *next=load32((byte*)n+l->size+4); // 必须先保存
        unlink(l,n);
        free_adapter(n);
        n=next;
    }
}
```

unlink完整范围 `a0089d02..9d75`：head分支先写head=next，无next清tail，否则清next.prev；非head先取prev，再区分tail或中间节点，tail更新tail=prev，必要清head/prev.next，中间依次写prev.next=next、next.prev=prev。unlink不free，free由walker调用adapter；adapter对NULL/sentinel早退，正常36 B节点经backend `a002e14e`归还，backend内部分配器不扩展。

已知初始PNG→BIN状态下，clear先保存BIN、unlink/free PNG，再unlink/free BIN，最终head/tail=0。此clear零callback分支不会调用已注册decoder.close。通用walker非零callback分支确实存在 `ae3e C.JALR s3`；它也不能被误认成读取node+c的decoder close。`bf44` 最后清G initialized byte，节点回收与open session关闭是不同合同。

状态为 `global未初始化 → 正常init/注册 → registry供同步info借用 → 进入global deinit → cache清理 → registry逐节点unlink/free → initialized=0`。registry拥有node，info仅借node和ROM descriptor/data/name/signature，local/header为同步借用，均无本路径所有权转移；不把返回node误当成调用者须free的对象。注册node驻留于global会话，weather图标更新不逐次分配/销毁它。

父函数片段重新核到 `cc78` 按G initialized为0跳 `cd36` 调init；`eea0` 读G+1，零时到 `eeaa` 保存返回值，`eeae` 调deinit、重载后跳公共尾部。因此全局清理存在真实上层调用，但完整父函数退出条件、所有调用者或所有导航是否触达它仍未知。
